const FOLDER_ID = "1m5jNNFrrIATNtlxcssvzLMU2XNXpj5oN"; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); let fileUrl = "Tidak ada foto"; // Jika ada bukti foto (format Base64 dari aplikasi) if (data.photoBase64 && data.photoBase64.includes(",")) { const folder = DriveApp.getFolderById(FOLDER_ID); const base64Data = data.photoBase64.split(",")[1]; const decoded = Utilities.base64Decode(base64Data); // Penamaan file otomatis agar rapi di folder komputer const fileName = `${data.tanggal}_${data.unit}_${data.petugasRM}.jpg`.replace(/\s+/g, "_"); const blob = Utilities.newBlob(decoded, "image/jpeg", fileName); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fileUrl = file.getUrl(); } // Simpan baris rekap ke Google Sheets sheet.appendRow([ new Date(), data.tanggal, data.unit, data.petugasRM, data.petugasUnit, data.totalPasien, data.berkasLengkap, data.itemKurang, fileUrl ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", photoUrl: fileUrl })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
